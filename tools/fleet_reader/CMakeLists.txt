cmake_minimum_required(VERSION 3.16)
project(living_fleet_reader CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(fleet_reader src/main.cpp src/sii.cpp src/fleet.cpp)
if(MSVC)
  target_compile_options(fleet_reader PRIVATE /W4 /utf-8)
  set_property(TARGET fleet_reader PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
enable_testing()
add_executable(fleet_tests tests/test_main.cpp src/sii.cpp src/fleet.cpp)
add_test(NAME fleet_tests COMMAND fleet_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/sample_game.sii)
